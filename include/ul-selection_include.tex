\providecommand{\ulMainBookTitle}{Unilaiva Selection}
\providecommand{\ulBookMotto}{a selection of songs from Unilaiva Songbook}
\providecommand{\fullbookcoverimage}{Unilaiva-songbook_COVER.pdf}
\providecommand{\ulBookWebsiteLinkPreText}{Find the complete songbook at:}

\ulUppercaseBookTitleInHeader{true}
\ulUppercaseSectionTitleInHeader{true}

% Do not show chapter nor section in the page headers, as they don't exist.
% Instead spread the book title and subtitle onto odd and even pages.
\fancypagestyle{unilaiva}[fancy]{%
  \fancyhead[CE]{%
    \ulHeaderTitleStyle%
    \ifulUppercaseBookTitleInHeader%
      \protect\MakeUppercase{\expandafter\ulMainBookTitle}%
    \else%
      \ulMainBookTitle%
    \fi%
  }
  \fancyhead[CO]{%
    \ulHeaderTitleStyle%
    \ifdefempty{\ulSubBookTitle}{}{% not empty
      \ifulUppercaseSectionTitleInHeader%
        \protect\MakeUppercase{\expandafter\ulSubBookTitle}%
      \else%
        \ulSubBookTitle%
      \fi%
    }%
  }
}

\ulSetPermanentCoverIcons[ulsbs-tag-icon-partial-edition_EN_512x512px.png]%
\ulSetVariableCoverIcons[ulsbs-tag-icon-chords_EN_512x512px]

\providecommand{\coverpdf}{}

% This will create the cover page for a selection booklet. If macro \coverpdf
% is not empty, it's content will be assumed to be a file name of a PDF file in
% content/img, and will be used as a cover page. Otherwise a cover will be
% generated from \ulMainBookTitle, \ulSubBookTitle (optional) and
% \fullbookcoverimage.
\newcommand{\coverpageforselection}{%
  \ifdefempty{\coverpdf}
    {% \coverpdf is empty
      {% must be contained within a block for the settings not to bleed
        \thispagestyle{empty}%
        \topskip0pt%
        \vspace*{11.2ex} % align with "Contents" heading on the next page; BAD for any changes
        \begin{center}
          {\Huge \ulMainBookTitle}
          \ifdefempty{\ulBookMotto}{}{% not empty
            {\par\normalsize\textit{\ulBookMotto}}
          }%
          \par\vspace*{6.98ex}
          \ifdefempty{\ulSubBookTitle}{}{% not empty
            {\Large \textbf{\ulSubBookTitle}}
          }%
          \par\vspace*{\fill}\vspace*{\fill}
          \imagec[2]{\fullbookcoverimage}
          \par\vspace*{\fill}
        \end{center}
        \ulIfVariant{lyric}{% skip given set ii icons in lyric-only books
          \ulSetVariableCoverIcons[\ulLyricsOnlyTagIcon]%
        }{}%
        \ulInsertCoverIconsHbc
      }
    }%
    {% \coverpdf not empty
      \ulCoverPage{\coverpdf}%
    }%
}
